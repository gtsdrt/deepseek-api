function downloadDrawio() { if (!authToken || isTokenExpired(authToken)) { clearSessionAndLogout(); return; } const subscriptionId = document.getElementById('subscriptionSelect').value; if (!subscriptionId) { alert('请先选择订阅'); return; } const resultDiv = document.getElementById('result'); resultDiv.innerHTML = '⏳ 正在生成 Draw.io 图表...'; fetch(`${BASE_URL}/export_drawio?subscription_id=${encodeURIComponent(subscriptionId)}`, { headers: { 'Authorization': `Bearer ${authToken}`, 'Ocp-Apim-Subscription-Key': APIM_SUBSCRIPTION_KEY } }) .then(response => { if (!response.ok) { return response.json().then(err => { throw new Error(err.error || `HTTP ${response.status}`); }); } const contentDisposition = response.headers.get('Content-Disposition'); let filename = 'azure_architecture.drawio'; if (contentDisposition) { const match = contentDisposition.match(/filename=(.+)/); if (match) filename = match[1]; } return response.blob().then(blob => ({ blob, filename })); }) .then(({ blob, filename }) => { const link = document.createElement('a'); link.href = URL.createObjectURL(blob); link.download = filename; document.body.appendChild(link); link.click(); document.body.removeChild(link); URL.revokeObjectURL(link.href); resultDiv.innerHTML = `✅ 图表已下载: ${filename}`; }) .catch(error => { resultDiv.innerHTML = `❌ 导出失败: ${error.message}`; }); }
